╔═══════════════════════════════════════════════════════════════╗
║           ПРАВИЛЬНЫЕ ЧАСТОТЫ ПРОФИЛЕЙ BMI30                  ║
╚═══════════════════════════════════════════════════════════════╝

PROFILE=1 → 176 Гц (≈176.0 Hz, не 200 Hz!)
PROFILE=2 → 280 Гц (≈280.0 Hz, не 300 Hz!)

Оба профиля используют:
- total_samples = 912
- Чередующиеся кадры A и B

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

ИСПОЛЬЗОВАНИЕ В КОДЕ:

# Инициализация для 176 Hz
stream = usb_stream.BMI30(profile=1, full=True)

# Инициализация для 280 Hz
stream = usb_stream.BMI30(profile=2, full=True)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

ТЕСТИРОВАНИЕ:

# Диагностика профиля 1 (176 Hz)
python3 host/diag_block_rate.py --profile 1 --duration 5

# Диагностика профиля 2 (280 Hz)
python3 host/diag_block_rate.py --profile 2 --duration 5

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

ВАЖНО:

Эти частоты - характеристика аппаратной части прошивки STM32H723.
Они не могут быть изменены командами хоста (SET_BLOCK_RATE и т.д.).

Если требуются точно 200 Hz и 300 Hz:
→ Необходимо обновление прошивки устройства
